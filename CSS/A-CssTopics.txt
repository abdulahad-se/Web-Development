18-Css Grid.(sourcefile: index18.html and index18part2.html).
    -Grid Container Properties.
    --------------------------------------------------------------------------
        -justify-content:;
        ----------------------------------------------------------------
        -align-content:;
        -----------------------------------------------------------
        -grid-tempelate-columns: e.g 100px 120px;
        ---------------------------------------------------------
        -grid-tempelate-rows: e.g 120px 210px;
        -------------------------------------------------------
        -Naming Grid Lines:
            -grid-tempelate-columns: e.g [first]90px [second]130px;
            -grid-tempelate-rows: e.g [one]80px [two]140px;
        ---------------------------------------------------------------
        -Grid tempelate Area:
            -e.g : grid-template-areas: "nav nav nav"
                                 "side article article"
                                 "footer footer footer";
            -e.g ( how i use in items):
                .item-1{
                    grid-area:nav;
                }
        ------------------------------------------------------------------------
        -fr units:
            -grid-tempelate-columns:1fr 1fr 1fr;
            -grid-tempelate-rows:1fr 2fr 1fr;
        ---------------------------------------------------------------------
        -repeat:
            -grid-tempelate-columns:repeat(4 , 1fr);
            -grid-tempelate-rows:repeat(3,1fr);
        -------------------------------------------------------------
        -minmax:
            -grid-tempelate-columns:repeat(4 ,minmax(120px,1fr));
            -grid-tempelate-rows:repeat(3 , minmax(100px,1fr));
        --------------------------------------------------------------------
        -gap:
            gap: e.g 20px;
            -row-gap: e.g 40px;
            -column-gap: e.g 30px;
        -justify-items:center | start | end | left | right | flex-start | flex-end etc;
        -align-items:center | start | end | stretch | baseline | flex-start | flex-end etc;
        -------------------------------------------------------------------------------

    Grid Items Properties.
    -------------------------------------
        -grid-column: e.g 1/2;
        -grid-column-start: e.g 1;
        -grid-column-end: e.g 2;
        -grid-column: span 2;
        ----------------------------------------
        -grid-row: e.g 1/3;
        -grid-row-start: e.g 1;
        -grid-row-end: e.g 3;
        -grid-row:span 3;
        -----------------------------------------------
        -justify-self:center etc;
        -align-self:center etc;
